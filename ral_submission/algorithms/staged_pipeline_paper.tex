% \parState: a numbered statement whose continuation lines keep the
% statement's indentation instead of returning to the left margin.
\makeatletter
\providecommand{\parState}[1]{\State\parbox[t]{\dimexpr\linewidth-\ALG@thistlm\relax}{\raggedright\strut#1\strut}}
\makeatother

\begin{algorithm}[!t]
\caption{Staged VLA adaptation}
\label{alg:paper_pipeline}
\small
\algrenewcommand\algorithmicindent{1.0em}
\begin{algorithmic}[1]
\Require Pretrained VLA, skill instructions $\ell_k$, success conditions $E_k$, full-task instruction $\ell$
\parState{Initialize policy $\pi$ with a new residual controller. Initialize one empty success bank $\mathcal B_k$ per skill}
\parState{Set GRPO options $\mathcal H$ with $G=8$, clip margins $(0.20,0.28)$, and dynamic sampling}
\For{reward type $\in$ (dense, binary)}
  \For{$k=1,\ldots,n$}
    \State Set skill reward $R_k$ and reset schedule $\mathcal S_k$
    \State $\pi\gets\Call{GRPO}{\pi,\ell_k,R_k,\mathcal S_k,\mathcal H}$
    \parState{Smooth and replay successful trajectories. Add replays that satisfy $E_k$ to $\mathcal B_k$}
    \State Evaluate all skills learned so far
    \If{a skill's success falls more than $\delta$ below its best rate}
      \parState{Fit a recovery policy on the banks with equal weight per skill}
      \parState{Adopt it if the lost skill recovers and the other skills retain their success}
    \EndIf
  \EndFor
\EndFor
\parState{Relabel the bank segments with $\ell$. Fit $\pi$ to the relabeled actions}
\State Set empty-gripper resets $\mathcal S$ and milestone rewards $R^{\mathrm{ms}}$
\State $\pi\gets\Call{GRPO}{\pi,\ell,R^{\mathrm{ms}},\mathcal S,\mathcal H}$
\If{success under $\ell$ stops improving}
  \State Freeze the VLA and its adapters
  \State $\pi\gets\Call{GRPO}{\pi,\ell,R^{\mathrm{ms}},\mathcal S,\mathcal H}$
\EndIf
\State \Return $\pi$
\end{algorithmic}
\end{algorithm}
